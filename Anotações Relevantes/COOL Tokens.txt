COOL Tokens
-----------------


-Keywords
       \
	`->  class, inherits, let, in, if, then, else, fi, while, 
             self, true, false, not, case, esac, isvoid
	     loop, pool, new

TOKEN: KEY_CLASS -> class ( Case insensitive )
TOKEN: KEY_INHERITS -> inherits ( Case insensitive )
TOKEN: KEY_LET -> let ( Case insensitive )
TOKEN: KEY_IN -> in ( Case insensitive )
TOKEN: KEY_IF -> if ( Case insensitive )
TOKEN: KEY_THEN -> then ( Case insensitive )
TOKEN: KEY_ELSE -> else ( Case insensitive )
TOKEN: KEY_FI -> fi ( Case insensitive )
TOKEN: KEY_WHILE -> while ( Case insensitive )
TOKEN: KEY_SELF -> self ( Case insensitive )
TOKEN: KEY_TRUE -> true ( Case Sensitive ) 
TOKEN: KEY_FALSE -> false ( Case Sensitive )
TOKEN: KEY_NOT -> not ( Case insensitive )
TOKEN: KEY_CASE -> case ( Case insensitive )
TOKEN: KEY_ESAC -> esac ( Case insensitive )
TOKEN: KEY_ISVOID -> isvoid ( Case insensitive )
TOKEN: KEY_LOOP -> loop ( Case insensitive )
TOKEN: KEY_POOL -> pool ( Case insensitive )
TOKEN: KEY_NEW -> new ( Case insensitive )


TOKEN: IDENTIFIERS -> [a-zA-Z]{1}[\w]*
	
ex: x, id, foo

  
TOKEN: NUM_LITERAL -> [0-9]+
	
ex: 2, 75, 92


TOKEN: STRING_LITERAL -> "[a-zA-Z0-9\n\t]*"
	
ex: "banana", "terra", "Area 51"


-Operators
	\
	 `->  =, +, -, *, /, <-

TOKEN: OP_EQUALS -> =
TOKEN: OP_PLUS -> +
TOKEN: OP_MINUS -> -
TOKEN: OP_TIMES -> *
TOKEN: OP_DIVIDE -> /
TOKEN: OP_ASSIGN -> <-


-Delimiters
	\
	 `->  :, ;, {, }, (, )

TOKEN: DELIM_COLON -> :
TOKEN: DELIM_SEMICOLON -> ;
TOKEN: DELIM_LBRACE -> {
TOKEN: DELIM_RBRACE -> }
TOKEN: DELIM_LPAREN -> (
TOKEN: DELIM_RPAREN -> )


-Whitespaces
	\
	 `->  \n, \t

TOKEN: WS_NEWLINE -> \n
TOKEN: WS_TAB -> \t


TOKEN: COMMENT -> \(\*[\w\s]*\*\)
